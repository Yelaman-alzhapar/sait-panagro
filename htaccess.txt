# panagro.kz — Apache (hoster.kz)
Options -Indexes
DirectoryIndex index.html
ErrorDocument 404 /404/index.html

<IfModule mod_rewrite.c>
RewriteEngine On
# https + без www
RewriteCond %{HTTPS} off [OR]
RewriteCond %{HTTP_HOST} ^www\. [NC]
RewriteRule ^(.*)$ https://panagro.kz/$1 [R=301,L]
# старые адреса Tilda (/page12345.html и т.п.) -> главная
RewriteRule ^page[0-9]+\.html$ / [R=301,L]
# слэш в конце для разделов
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_URI} !/$
RewriteCond %{REQUEST_FILENAME}/ -d
RewriteRule ^(.*)$ /$1/ [R=301,L]
</IfModule>
# Старые ссылки вида /?utm=... ведут на главную; canonical убирает дубли из индекса.
# Точное соответствие старых ?utm= новым разделам добавим после выгрузки списка из Search Console.

<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/css application/javascript image/svg+xml application/xml text/plain
</IfModule>
<IfModule mod_expires.c>
ExpiresActive On
ExpiresByType image/webp "access plus 1 year"
ExpiresByType image/png "access plus 1 year"
ExpiresByType image/svg+xml "access plus 1 year"
ExpiresByType text/css "access plus 1 month"
ExpiresByType application/javascript "access plus 1 month"
ExpiresByType text/html "access plus 1 hour"
</IfModule>
